const x = 5
function add(value){
	value++;
	return value
}

console.log(add(x), x)
6, 5

const obj2 = { count: 5 };
function add2(val){
	value.count++;
	return value.count;
}

console.log(add2(obj2), obj2.count)
6, 6 
Because in the second one you are passing by reference and not by value, changing the original object that the reference is pointing to
